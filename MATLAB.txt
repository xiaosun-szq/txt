Silmulink自定义库
    命令
    set_param(gcs,'EnableLBRepository','on');

    m脚本
    function blkStruct = slblocks
    % This function specifies that the library 'mylib'
    % should appear in the Library Browser with the
    % name 'My Library'

    Browser.Library = 'test';
    % 'mylib' is the name of the library

    Browser.Name = 'My_Library';
    % 'My Library' is the library name that appears
    % in the Library Browser

    blkStruct.Browser = Browser;

    m脚本
    function sl_customization(cm)
    % Change the order of libraries in the Simulink Library Browser.
    cm.LibraryBrowserCustomizer.applyOrder({'My_Library',-2});
    end

    命令
    sl_refresh_customizations



我们也可以使用 m 脚本来创建 Simulink.Parameter

m = Simulink.Parameter;
m.Value = 1200;
m.Complexity = 'real';
m.CoderInfo.StorageClass = 'Auto';
m.Description = '质量';
m.DataType = 'uint16';
m.Min = [];
m.Max = [];
m.DocUnits = 'kg';


我们也可以使用 m 脚本来创建 Simulink.Bus

VehicleInfo = Simulink.Bus;
VehicleInfo.Description = 'vehicle information';
VehicleInfo.DataScope = 'Auto';
% VehicleInfo.HeaderFile = 'yes.h';
VehicleInfo.Alignment = -1;
VehicleInfo.PreserveElementDimensions = false;

BusElements(1) = Simulink.BusElement;
BusElements(1).Name = 'mass';

BusElements(1).Dimensions = 1;
BusElements(1).DataType = 'int16';

BusElements(1).Max = [];
BusElements(1).DimensionsMode = 'Fixed';
BusElements(1).SamplingMode = 'Sample based';
BusElements(1).DocUnits = 'kg';

BusElements(2) = Simulink.BusElement;

BusElements(2).Max = [];
